package main

import (
	"context"
	"errors"
	"log"
	"net/http"
	"os"
	"os/signal"
	"time"

	"expense-api/handlers"
	"expense-api/repository"
	"expense-api/service"
)

func main() {
	// Build the layers from the bottom up and connect them.
	repo := repository.NewInMemoryExpenseRepository()
	svc := service.NewExpenseService(repo)
	handler := handlers.NewExpenseHandler(svc)

	mux := http.NewServeMux()
	handler.RegisterRoutes(mux)

	// Timeouts stop slow or stuck clients from holding connections open forever.
	// LogRequests wraps the router so every request is printed to the terminal.
	server := &http.Server{
		Addr:              ":8080",
		Handler:           handlers.LogRequests(mux),
		ReadHeaderTimeout: 5 * time.Second,
		ReadTimeout:       10 * time.Second,
		WriteTimeout:      10 * time.Second,
		IdleTimeout:       60 * time.Second,
	}

	// Start the server in its own goroutine so main can wait for events.
	// The buffered channel carries any start-up error back to main.
	serverErr := make(chan error, 1)
	go func() {
		log.Println("Expense API listening on http://localhost:8080")
		serverErr <- server.ListenAndServe()
	}()

	// Ctrl+C is an operating-system event. signal.NotifyContext turns it into
	// a context that is cancelled when the signal arrives.
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()

	// select waits for whichever event happens first.
	select {
	case err := <-serverErr:
		if !errors.Is(err, http.ErrServerClosed) {
			log.Fatalf("server error: %v", err)
		}
	case <-ctx.Done():
		log.Println("Shutting down: finishing requests already in progress...")
		// Give running requests up to 5 seconds to finish.
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		if err := server.Shutdown(shutdownCtx); err != nil {
			log.Printf("shutdown error: %v", err)
		}
		log.Println("Server stopped")
	}
}

package models

// Expense represents one spending record made by a student.
type Expense struct {
	ID       int     `json:"id"`       // unique number identifying the expense
	Title    string  `json:"title"`    // short name, e.g. "Lunch"
	Amount   float64 `json:"amount"`   // money spent (float64 is simple, but can have rounding errors)
	Category string  `json:"category"` // grouping used for the summary, e.g. "Food"
	Date     string  `json:"date"`     // stored as text (YYYY-MM-DD) to keep the project simple
	Note     string  `json:"note"`     // optional extra detail
}

package repository

import (
	"sort"
	"sync"

	"expense-api/models"
)

// ExpenseRepository describes how expenses are stored. Any type that has
// these methods satisfies the interface automatically (implicit interfaces).
type ExpenseRepository interface {
	Create(e models.Expense) models.Expense
	GetAll() []models.Expense
	GetByID(id int) (models.Expense, bool)
	Update(id int, e models.Expense) (models.Expense, bool)
	Delete(id int) bool
}

// InMemoryExpenseRepository keeps expenses in a map held in memory.
// The mutex makes it safe when several requests arrive at the same time.
type InMemoryExpenseRepository struct {
	mu       sync.RWMutex
	expenses map[int]models.Expense
	nextID   int
}

// This line does nothing at run time. It makes the compiler check that
// InMemoryExpenseRepository has every method of ExpenseRepository, even
// though Go never needs an "implements" keyword.
var _ ExpenseRepository = (*InMemoryExpenseRepository)(nil)

// NewInMemoryExpenseRepository creates an empty repository.
func NewInMemoryExpenseRepository() *InMemoryExpenseRepository {
	return &InMemoryExpenseRepository{
		expenses: make(map[int]models.Expense),
		nextID:   1,
	}
}

// Create assigns a new ID, stores the expense and returns it.
func (r *InMemoryExpenseRepository) Create(e models.Expense) models.Expense {
	r.mu.Lock()
	defer r.mu.Unlock() // unlocks when the function ends
	e.ID = r.nextID
	r.nextID++
	r.expenses[e.ID] = e
	return e
}

// GetAll returns every expense, sorted by ID.
func (r *InMemoryExpenseRepository) GetAll() []models.Expense {
	r.mu.RLock()
	defer r.mu.RUnlock()
	list := make([]models.Expense, 0, len(r.expenses))
	for _, e := range r.expenses {
		list = append(list, e)
	}
	sort.Slice(list, func(i, j int) bool { return list[i].ID < list[j].ID })
	return list
}

// GetByID returns one expense; the bool says whether it was found.
func (r *InMemoryExpenseRepository) GetByID(id int) (models.Expense, bool) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	e, ok := r.expenses[id]
	return e, ok
}

// Update replaces an existing expense, keeping its ID.
func (r *InMemoryExpenseRepository) Update(id int, e models.Expense) (models.Expense, bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if _, ok := r.expenses[id]; !ok {
		return models.Expense{}, false
	}
	e.ID = id
	r.expenses[id] = e
	return e, true
}

// Delete removes an expense and reports whether it existed.
func (r *InMemoryExpenseRepository) Delete(id int) bool {
	r.mu.Lock()
	defer r.mu.Unlock()
	if _, ok := r.expenses[id]; !ok {
		return false
	}
	delete(r.expenses, id)
	return true
}

package service

import (
	"errors"
	"fmt"
	"math"
	"strings"
	"time"

	"expense-api/models"
	"expense-api/repository"
)

// Custom errors. Go returns errors as values instead of throwing exceptions.
var (
	ErrNotFound       = errors.New("expense not found")
	ErrInvalidExpense = errors.New("invalid expense")
	ErrInvalidFilter  = errors.New("invalid filter")
)

// Filter limits which expenses are listed or summarised. Empty fields are
// ignored, so an empty Filter matches every expense.
type Filter struct {
	Category string // only this category (upper/lower case does not matter)
	From     string // earliest date to include, YYYY-MM-DD
	To       string // latest date to include, YYYY-MM-DD
}

// Summary holds the totals returned by the /summary endpoint.
// Category, From and To show which filter was used, and are left out of the
// JSON when empty.
type Summary struct {
	Category    string             `json:"category,omitempty"`
	From        string             `json:"from,omitempty"`
	To          string             `json:"to,omitempty"`
	Count       int                `json:"count"`
	TotalAmount float64            `json:"total_amount"`
	ByCategory  map[string]float64 `json:"by_category"`
}

// ExpenseService contains the business rules. It depends on the
// repository interface, not on a specific storage type.
type ExpenseService struct {
	repo repository.ExpenseRepository
}

// NewExpenseService creates a service using the given repository.
func NewExpenseService(repo repository.ExpenseRepository) *ExpenseService {
	return &ExpenseService{repo: repo}
}

// clean trims spaces from the text fields, so " Food " and "Food" are
// treated as the same category.
func clean(e models.Expense) models.Expense {
	e.Title = strings.TrimSpace(e.Title)
	e.Category = strings.TrimSpace(e.Category)
	e.Date = strings.TrimSpace(e.Date)
	e.Note = strings.TrimSpace(e.Note)
	return e
}

// validate checks the rules for an expense. Each error is wrapped with %w,
// so callers can test it with errors.Is(err, ErrInvalidExpense).
func validate(e models.Expense) error {
	if e.Title == "" {
		return fmt.Errorf("%w: title must not be empty", ErrInvalidExpense)
	}
	// Bad aspect (zero values): if "amount" is missing from the JSON, Go sets it
	// to 0, so a missing amount and a real 0 look the same. This check catches both.
	if e.Amount <= 0 {
		return fmt.Errorf("%w: amount must be positive", ErrInvalidExpense)
	}
	if e.Category == "" {
		return fmt.Errorf("%w: category must not be empty", ErrInvalidExpense)
	}
	// The date is optional, but if given it must be a real YYYY-MM-DD date.
	if e.Date != "" && !isDate(e.Date) {
		return fmt.Errorf("%w: date must be in YYYY-MM-DD format", ErrInvalidExpense)
	}
	return nil
}

// isDate reports whether s is a real date in YYYY-MM-DD format.
func isDate(s string) bool {
	_, err := time.Parse(time.DateOnly, s)
	return err == nil
}

// cleaned returns a copy of the filter with spaces trimmed from each field.
func (f Filter) cleaned() Filter {
	f.Category = strings.TrimSpace(f.Category)
	f.From = strings.TrimSpace(f.From)
	f.To = strings.TrimSpace(f.To)
	return f
}

// check reports an error if a date is badly formatted or the range is backwards.
func (f Filter) check() error {
	if f.From != "" && !isDate(f.From) {
		return fmt.Errorf("%w: from must be in YYYY-MM-DD format", ErrInvalidFilter)
	}
	if f.To != "" && !isDate(f.To) {
		return fmt.Errorf("%w: to must be in YYYY-MM-DD format", ErrInvalidFilter)
	}
	if f.From != "" && f.To != "" && f.From > f.To {
		return fmt.Errorf("%w: from must not be after to", ErrInvalidFilter)
	}
	return nil
}

// matches reports whether an expense passes the filter. Dates in YYYY-MM-DD
// format sort the same way as text, so they can be compared as strings.
func (f Filter) matches(e models.Expense) bool {
	if f.Category != "" && !strings.EqualFold(e.Category, f.Category) {
		return false
	}
	if (f.From != "" || f.To != "") && e.Date == "" {
		return false // an expense without a date cannot be inside a date range
	}
	if f.From != "" && e.Date < f.From {
		return false
	}
	if f.To != "" && e.Date > f.To {
		return false
	}
	return true
}

// filter checks f and returns only the expenses that match it.
func (s *ExpenseService) filter(f Filter) ([]models.Expense, error) {
	if err := f.check(); err != nil {
		return nil, err
	}
	list := make([]models.Expense, 0)
	for _, e := range s.repo.GetAll() {
		if f.matches(e) {
			list = append(list, e)
		}
	}
	return list, nil
}

// roundCents rounds an amount to 2 decimal places.
// Bad aspect (no decimal type): float64 cannot store most decimals exactly, so
// 0.1 + 0.2 gives 0.30000000000000004. Totals are rounded before being returned.
func roundCents(amount float64) float64 {
	return math.Round(amount*100) / 100
}

// Add cleans, validates and stores a new expense.
func (s *ExpenseService) Add(e models.Expense) (models.Expense, error) {
	e = clean(e)
	if err := validate(e); err != nil {
		return models.Expense{}, err
	}
	return s.repo.Create(e), nil
}

// List returns the expenses that match the filter, sorted by ID.
// An empty Filter returns every expense.
func (s *ExpenseService) List(f Filter) ([]models.Expense, error) {
	return s.filter(f.cleaned())
}

// Get returns one expense or ErrNotFound.
func (s *ExpenseService) Get(id int) (models.Expense, error) {
	e, ok := s.repo.GetByID(id)
	if !ok {
		return models.Expense{}, ErrNotFound
	}
	return e, nil
}

// Update cleans, validates and replaces an expense, or returns ErrNotFound.
func (s *ExpenseService) Update(id int, e models.Expense) (models.Expense, error) {
	e = clean(e)
	if err := validate(e); err != nil {
		return models.Expense{}, err
	}
	updated, ok := s.repo.Update(id, e)
	if !ok {
		return models.Expense{}, ErrNotFound
	}
	return updated, nil
}

// Delete removes an expense, or returns ErrNotFound.
func (s *ExpenseService) Delete(id int) error {
	if !s.repo.Delete(id) {
		return ErrNotFound
	}
	return nil
}

// Summary adds up the total spending and the total per category for the
// expenses that match the filter, for example one month of spending.
func (s *ExpenseService) Summary(f Filter) (Summary, error) {
	f = f.cleaned()
	list, err := s.filter(f)
	if err != nil {
		return Summary{}, err
	}
	sum := Summary{
		Category:   f.Category,
		From:       f.From,
		To:         f.To,
		ByCategory: make(map[string]float64),
	}
	for _, e := range list {
		sum.Count++
		sum.TotalAmount += e.Amount
		sum.ByCategory[e.Category] += e.Amount
	}
	sum.TotalAmount = roundCents(sum.TotalAmount)
	for category, total := range sum.ByCategory {
		sum.ByCategory[category] = roundCents(total)
	}
	return sum, nil
}

package handlers

import (
	"encoding/json"
	"errors"
	"fmt"
	"net/http"
	"strconv"

	"expense-api/models"
	"expense-api/service"
)

// ExpenseHandler turns HTTP requests into service calls.
type ExpenseHandler struct {
	svc *service.ExpenseService
}

// NewExpenseHandler creates a handler using the given service.
func NewExpenseHandler(svc *service.ExpenseService) *ExpenseHandler {
	return &ExpenseHandler{svc: svc}
}

// RegisterRoutes connects each URL pattern to its handler function.
// Since Go 1.22 a pattern holds both the HTTP method and the path, and
// {id} marks a path variable that r.PathValue("id") reads.
func (h *ExpenseHandler) RegisterRoutes(mux *http.ServeMux) {
	mux.HandleFunc("POST /expenses", h.create)
	mux.HandleFunc("GET /expenses", h.list)
	mux.HandleFunc("GET /expenses/{id}", h.get)
	mux.HandleFunc("PUT /expenses/{id}", h.update)
	mux.HandleFunc("DELETE /expenses/{id}", h.delete)
	mux.HandleFunc("GET /summary", h.summary)
}

// writeJSON sends a JSON response with the given status code.
func writeJSON(w http.ResponseWriter, status int, data any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(data)
}

// writeError converts an error into a status code and a JSON message.
func writeError(w http.ResponseWriter, err error) {
	status := http.StatusInternalServerError
	switch {
	case errors.Is(err, service.ErrNotFound):
		status = http.StatusNotFound
	case errors.Is(err, service.ErrInvalidExpense), errors.Is(err, service.ErrInvalidFilter):
		status = http.StatusBadRequest
	}
	writeJSON(w, status, map[string]string{"error": err.Error()})
}

// parseID reads the {id} part of the URL as a number.
func parseID(r *http.Request) (int, error) {
	id, err := strconv.Atoi(r.PathValue("id"))
	if err != nil {
		return 0, fmt.Errorf("%w: id must be a number", service.ErrInvalidExpense)
	}
	return id, nil
}

// filterFromQuery reads the optional query parameters, for example
// /expenses?category=Food&from=2026-10-01&to=2026-10-31.
func filterFromQuery(r *http.Request) service.Filter {
	q := r.URL.Query()
	return service.Filter{
		Category: q.Get("category"),
		From:     q.Get("from"),
		To:       q.Get("to"),
	}
}

// maxBodyBytes limits how large a request body can be (1 MB).
const maxBodyBytes = 1 << 20

// decodeExpense reads an Expense from the JSON request body. Unknown fields
// (for example a misspelled "amout") are rejected instead of being ignored.
func decodeExpense(w http.ResponseWriter, r *http.Request) (models.Expense, error) {
	var e models.Expense
	dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBodyBytes))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&e); err != nil {
		return e, fmt.Errorf("%w: body must be valid JSON", service.ErrInvalidExpense)
	}
	return e, nil
}

// create handles POST /expenses: it adds an expense and returns it with 201 Created.
// Bad aspect (repetitive error handling): every call that can fail needs its own
// "if err != nil" block. writeError shortens each block, but cannot remove them.
func (h *ExpenseHandler) create(w http.ResponseWriter, r *http.Request) {
	e, err := decodeExpense(w, r)
	if err != nil {
		writeError(w, err)
		return
	}
	created, err := h.svc.Add(e)
	if err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, http.StatusCreated, created)
}

// list handles GET /expenses: it returns every expense sorted by ID, or only
// the ones matching ?category=, ?from= and ?to= when they are given.
func (h *ExpenseHandler) list(w http.ResponseWriter, r *http.Request) {
	expenses, err := h.svc.List(filterFromQuery(r))
	if err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, expenses)
}

// get handles GET /expenses/{id}: it returns one expense, or 404 if it does not exist.
func (h *ExpenseHandler) get(w http.ResponseWriter, r *http.Request) {
	id, err := parseID(r)
	if err != nil {
		writeError(w, err)
		return
	}
	e, err := h.svc.Get(id)
	if err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, e)
}

// update handles PUT /expenses/{id}: it replaces an expense and returns the new version.
func (h *ExpenseHandler) update(w http.ResponseWriter, r *http.Request) {
	id, err := parseID(r)
	if err != nil {
		writeError(w, err)
		return
	}
	e, err := decodeExpense(w, r)
	if err != nil {
		writeError(w, err)
		return
	}
	updated, err := h.svc.Update(id, e)
	if err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, updated)
}

// delete handles DELETE /expenses/{id}: it removes an expense and returns 204 No Content.
func (h *ExpenseHandler) delete(w http.ResponseWriter, r *http.Request) {
	id, err := parseID(r)
	if err != nil {
		writeError(w, err)
		return
	}
	if err := h.svc.Delete(id); err != nil {
		writeError(w, err)
		return
	}
	w.WriteHeader(http.StatusNoContent)
}

// summary handles GET /summary: it returns the count, the total and the total
// per category. The same query parameters as list can limit it, for example
// to one month.
func (h *ExpenseHandler) summary(w http.ResponseWriter, r *http.Request) {
	sum, err := h.svc.Summary(filterFromQuery(r))
	if err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, sum)
}

package handlers

import (
	"log"
	"net/http"
	"time"
)

// statusRecorder wraps http.ResponseWriter so the status code can be read
// after the handler has run. Embedding the interface promotes all of its
// methods, so only WriteHeader needs to be written here (composition, not
// inheritance).
type statusRecorder struct {
	http.ResponseWriter
	status int
}

// WriteHeader remembers the status code, then passes it on.
func (r *statusRecorder) WriteHeader(status int) {
	r.status = status
	r.ResponseWriter.WriteHeader(status)
}

// LogRequests is middleware: a function that takes a handler and returns a
// new handler. The returned function is a closure that can still use next.
// It logs the method, path, status code and time taken for every request.
func LogRequests(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := time.Now()
		rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
		next.ServeHTTP(rec, r)
		log.Printf("%s %s -> %d (%v)", r.Method, r.URL.Path, rec.status, time.Since(start))
	})
}

package repository

import (
	"sync"
	"testing"

	"expense-api/models"
)

// TestConcurrentCreate starts many goroutines that add expenses at the same
// time, like many HTTP requests arriving together. The mutex must make sure
// no expense is lost and every ID is different.
func TestConcurrentCreate(t *testing.T) {
	repo := NewInMemoryExpenseRepository()
	const workers = 100

	var wg sync.WaitGroup
	for range workers {
		wg.Add(1)
		go func() {
			defer wg.Done()
			repo.Create(models.Expense{Title: "Lunch", Amount: 5, Category: "Food"})
		}()
	}
	wg.Wait() // block until every goroutine has finished

	all := repo.GetAll()
	if len(all) != workers {
		t.Fatalf("expected %d expenses, got %d", workers, len(all))
	}
	for i, e := range all {
		if e.ID != i+1 {
			t.Fatalf("expected IDs 1..%d in order, found ID %d at position %d", workers, e.ID, i)
		}
	}
}

package service

	"errors"

func newTestService() *ExpenseService {
	return NewExpenseService(repository.NewInMemoryExpenseRepository())
}

// mustAdd adds an expense and stops the test if it fails.
func mustAdd(t *testing.T, svc *ExpenseService, e models.Expense) models.Expense {
	t.Helper()
	created, err := svc.Add(e)
	if err != nil {
		t.Fatalf("Add(%+v) failed: %v", e, err)
	}
	return created
}

func TestAddRejectsEmptyTitle(t *testing.T) {
	svc := newTestService()
	_, err := svc.Add(models.Expense{Title: "", Amount: 5, Category: "Food"})
	if !errors.Is(err, ErrInvalidExpense) {
		t.Errorf("expected ErrInvalidExpense, got %v", err)
	}
}

func TestGetMissingExpense(t *testing.T) {
	svc := newTestService()

		t.Errorf("expected ErrNotFound, got %v", err)
	}

func TestSummaryTotals(t *testing.T) {

// newTestServer builds the full stack with an empty in-memory repository.
func newTestServer() *http.ServeMux {
	svc := service.NewExpenseService(repository.NewInMemoryExpenseRepository())
	mux := http.NewServeMux()
	NewExpenseHandler(svc).RegisterRoutes(mux)
	return mux
}

// do sends one request to the server and returns the recorded response.
func do(mux *http.ServeMux, method, path, body string) *httptest.ResponseRecorder {
	req := httptest.NewRequest(method, path, strings.NewReader(body))
	rec := httptest.NewRecorder()
	mux.ServeHTTP(rec, req)
	return rec
}

func TestStatusCodes(t *testing.T) {
	mux := newTestServer()
	valid := `{"title":"Lunch","amount":5,"category":"Food","date":"2026-10-12"}`

	tests := []struct {
		name         string
		method, path string
		body         string
		want         int
	}{
		{"create valid", "POST", "/expenses", valid, http.StatusCreated},
		{"create invalid JSON", "POST", "/expenses", `{`, http.StatusBadRequest},
		{"create unknown field", "POST", "/expenses", `{"title":"Lunch","amout":5,"category":"Food"}`, http.StatusBadRequest},
		{"create bad date", "POST", "/expenses", `{"title":"Lunch","amount":5,"category":"Food","date":"tomorrow"}`, http.StatusBadRequest},
		{"list", "GET", "/expenses", "", http.StatusOK},
		{"get existing", "GET", "/expenses/1", "", http.StatusOK},
		{"get missing", "GET", "/expenses/99", "", http.StatusNotFound},
		{"get non-numeric id", "GET", "/expenses/abc", "", http.StatusBadRequest},
		{"update existing", "PUT", "/expenses/1", valid, http.StatusOK},
		{"update missing", "PUT", "/expenses/99", valid, http.StatusNotFound},
		{"summary", "GET", "/summary", "", http.StatusOK},
		{"delete existing", "DELETE", "/expenses/1", "", http.StatusNoContent},
		{"delete again", "DELETE", "/expenses/1", "", http.StatusNotFound},
	}
	// The cases run in order and share one server, so later cases see earlier changes.
	for _, tt := range tests {
		rec := do(mux, tt.method, tt.path, tt.body)
		if rec.Code != tt.want {
			t.Errorf("%s: %s %s returned %d, want %d (body: %s)",
				tt.name, tt.method, tt.path, rec.Code, tt.want, rec.Body.String())
		}
	}
}

func TestQueryFilters(t *testing.T) {
	mux := newTestServer()
	do(mux, "POST", "/expenses", `{"title":"Lunch","amount":10,"category":"Food","date":"2026-09-28"}`)
	do(mux, "POST", "/expenses", `{"title":"Bus","amount":3,"category":"Transport","date":"2026-10-01"}`)
	do(mux, "POST", "/expenses", `{"title":"Dinner","amount":12,"category":"Food","date":"2026-10-15"}`)

	tests := []struct {
		name     string
		path     string
		wantCode int
		wantBody string // a piece of text the response must contain
	}{
		{"list by category", "/expenses?category=food", http.StatusOK, `"title":"Dinner"`},
		{"list by month", "/expenses?from=2026-10-01&to=2026-10-31", http.StatusOK, `"title":"Bus"`},
		{"summary by month", "/summary?from=2026-10-01&to=2026-10-31", http.StatusOK, `"total_amount":15`},
		{"bad date", "/expenses?from=tomorrow", http.StatusBadRequest, `from must be in YYYY-MM-DD format`},
		{"backwards range", "/summary?from=2026-10-31&to=2026-10-01", http.StatusBadRequest, `from must not be after to`},
	}
	for _, tt := range tests {
		rec := do(mux, "GET", tt.path, "")
		if rec.Code != tt.wantCode || !strings.Contains(rec.Body.String(), tt.wantBody) {
			t.Errorf("%s: GET %s returned %d %s, want %d containing %s",
				tt.name, tt.path, rec.Code, rec.Body.String(), tt.wantCode, tt.wantBody)
		}
	}

	// The September lunch must not appear in the October list.
	if body := do(mux, "GET", "/expenses?from=2026-10-01", "").Body.String(); strings.Contains(body, "Lunch") {
		t.Errorf("October list should not contain the September lunch: %s", body)
	}
}

package main

import "fmt"

func main() {
	count, total := 2, 12.5 // := declares two variables and infers int and float64
	const maxItems = 100    // a constant
	a, b := 1, 2
	a, b = b, a                   // multiple assignment: swap
	avg := total / float64(count) // explicit conversion; total / count would not compile

	fmt.Println("count:", count, "total:", total, "maxItems:", maxItems)
	fmt.Println("after swap a:", a, "b:", b)
	fmt.Println("average:", avg)
}

// Listing 8: starting goroutines and sending data over a channel.
// Run with: go run ./scratch/concurrency
package main

import (
	"fmt"
	"sync"
)

func main() {
	results := make(chan int) // unbuffered channel of ints
	var wg sync.WaitGroup
	for i := 1; i <= 3; i++ {
		wg.Add(1)
		go func(n int) { // "go" starts a goroutine
			defer wg.Done()
			results <- n * n // send blocks until main receives
		}(i)
	}
	go func() { wg.Wait(); close(results) }() // close the channel when all workers finish
	for r := range results {                  // receive until the channel is closed
		fmt.Println(r)
	}
}
